cmake_minimum_required(VERSION 3.28)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES OR CMAKE_CUDA_ARCHITECTURES STREQUAL "")
  message(FATAL_ERROR "Set CMAKE_CUDA_ARCHITECTURES explicitly for the target GPU")
endif()

project(cuda_component VERSION 0.1.0 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_EXTENSIONS OFF)

include(CTest)
option(CUDA_ENABLE_HOST_COVERAGE "Instrument host code for coverage" OFF)
option(CUDA_ENABLE_MUTATION "Register the required mutation suite" ON)
if(CUDA_ENABLE_HOST_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Host coverage is configured for GCC or Clang")
  endif()
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:--coverage>"
                      "$<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=--coverage>")
  add_link_options(--coverage)
endif()

find_package(CUDAToolkit REQUIRED)
add_library(cuda_component src/cuda/scale.cu)
target_include_directories(cuda_component PUBLIC include)
target_link_libraries(cuda_component PUBLIC CUDA::cudart)
target_compile_options(cuda_component PRIVATE
  "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>")
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
